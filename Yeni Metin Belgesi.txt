create database sec3;
go
use sec3;
go
create table Date_Dimension(
day_number int primary key,
day_date varchar(20),
week_number varchar(30),
month_number varchar(20),
year_number varchar(30));


create table Customer_Dimension(
customer_id int primary key,
customer_name  varchar(30),
customer_address  varchar(30),
other_details  varchar(30));

create table Product_Dimension(
product_code int primary key,
product_name varchar(20),
product_description varchar(20),
other_product_details varchar(30));

create table Geographic_Dimension(
geographic_area int primary key,
geographic_area_name varchar(20),
region varchar(20),
country varchar(30),
other_region_details varchar(30));
